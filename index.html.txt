    <script>
        // Expanded Database mapping concepts from past tests & Easy Notes PDF
        const quizData = [
            {
                module: "Language Acquisition",
                question: "Noam Chomsky uses the phrase 'Poverty of the Stimulus' to support his Innateness Theory by asserting that:",
                options: [
                    "Children rely completely on environmental rewards to correct grammatical anomalies.",
                    "Linguistic inputs in a child's surroundings are too limited to explain their rapid, structured language acquisition.",
                    "Children mimic language sounds without mapping their structural underlying meaning.",
                    "The Language Acquisition Device (LAD) only functions during intensive formal teaching."
                ],
                correct: 1,
                explanation: "The 'Poverty of the Stimulus' argument highlights that children are not exposed to enough linguistic variation to learn complex syntax via behavioral conditioning alone, demonstrating an innate grammatical blueprint (Universal Grammar)."
            },
            {
                module: "Intelligence Frameworks",
                question: "In J.P. Guilford's Structure of Intellect (SOI) Cubical Model, into which specific dimension are 'Divergent Thinking' and 'Evaluation' categorized?",
                options: ["Contents", "Products", "Operations", "Systems"],
                correct: 2,
                explanation: "Operations represent 'how the mind works' or the active processing steps. Guilford identified 6 unique Operations: Cognition, Memory Recording, Memory Retention, Divergent Thinking, Convergent Thinking, and Evaluation."
            },
            {
                module: "Memory Architectures",
                question: "According to the Atkinson-Shiffrin Stage Model, what structural mechanism causes the loss of item fields within Short-Term Memory (STM)?",
                options: [
                    "Trace decay and displacement",
                    "A semantic storage schema breakdown",
                    "Proactive cognitive retrieval blockade",
                    "Deficient contextual retrieval signals"
                ],
                correct: 0,
                explanation: "Atkinson & Shiffrin's framework states that STM has a finite duration (15–30 seconds) and structural capacity limits. Information degrades automatically over time (decay) or is forced out by incoming data (displacement)."
            },
            {
                module: "Problem Solving",
                question: "If an individual struggles to solve a novel dilemma because they persistently apply an operational method that proved effective in past scenarios, they are facing which cognitive barrier?",
                options: ["Functional Fixedness", "Mental Set", "Retroactive Inhibition", "Confirmation Bias"],
                correct: 1,
                explanation: "A Mental Set is an operational tendency to approach situations using frameworks or rules that worked previously, creating an obstacle to finding alternative, more effective problem-solving paths."
            },
            {
                module: "Intelligence Frameworks",
                question: "Under Guilford's SOI framework, if you arrange isolated concepts or data points into a structural, interrelated cluster, which 'Product' dimension have you developed?",
                options: ["Units", "Classes", "Systems", "Transformations"],
                correct: 2,
                explanation: "Products represent how information is structured or organized by our mental operations. 'Systems' specifically denote organized aggregates or structural sets of items working together."
            }
        ];

        let currentIndex = 0;
        let streak = 0;

        const moduleTag = document.getElementById('module-tag');
        const questionText = document.getElementById('question-text');
        const optionsContainer = document.getElementById('options-container');
        const feedbackBox = document.getElementById('feedback-box');
        const nextBtn = document.getElementById('next-btn');
        const progressBar = document.getElementById('progress');
        const streakBadge = document.getElementById('streak-badge');

        function loadQuestion() {
            feedbackBox.classList.add('hidden');
            nextBtn.classList.add('hidden');
            
            const currentItem = quizData[currentIndex];
            moduleTag.innerText = currentItem.module;
            questionText.innerText = currentItem.question;
            optionsContainer.innerHTML = '';

            progressBar.style.width = `${(currentIndex / quizData.length) * 100}%`;

            currentItem.options.forEach((option, index) => {
                const btn = document.createElement('button');
                btn.className = "w-full text-left p-4 border border-slate-200 rounded-xl hover:border-blue-500 hover:bg-blue-50/40 transition-all font-medium text-slate-700 text-xs sm:text-sm";
                btn.innerText = option;
                btn.onclick = () => verifySelection(index, btn);
                optionsContainer.appendChild(btn);
            });
        }

        function verifySelection(selectedIndex, selectedBtn) {
            const currentItem = quizData[currentIndex];
            const buttons = optionsContainer.getElementsByTagName('button');

            for (let btn of buttons) { btn.disabled = true; }

            if (selectedIndex === currentItem.correct) {
                selectedBtn.className = "w-full text-left p-4 border border-emerald-500 bg-emerald-50 rounded-xl text-emerald-800 font-semibold text-xs sm:text-sm";
                feedbackBox.className = "p-4 bg-emerald-50 border border-emerald-200 text-emerald-800 rounded-xl mb-4 text-xs font-medium";
                feedbackBox.innerHTML = `<strong>✨ Concept Mastered!</strong><br>${currentItem.explanation}`;
                streak++;
            } else {
                selectedBtn.className = "w-full text-left p-4 border border-rose-400 bg-rose-50 rounded-xl text-rose-800 font-medium text-xs sm:text-sm";
                buttons[currentItem.correct].className = "w-full text-left p-4 border border-emerald-500 bg-emerald-50 rounded-xl text-emerald-800 font-semibold text-xs sm:text-sm";
                feedbackBox.className = "p-4 bg-rose-50 border border-rose-200 text-rose-800 rounded-xl mb-4 text-xs font-medium";
                feedbackBox.innerHTML = `<strong>💡 Core Insight Review:</strong><br>${currentItem.explanation}`;
                streak = 0;
            }

            streakBadge.innerText = `🔥 Streak: ${streak}`;
            feedbackBox.classList.remove('hidden');
            nextBtn.classList.remove('hidden');
        }

        nextBtn.onclick = () => {
            currentIndex++;
            if (currentIndex < quizData.length) {
                loadQuestion();
            } else {
                currentIndex = 0;
                progressBar.style.width = "100%";
                alert("🎉 Structural Framework Retained! Resetting module pathways to secure long-term active recall.");
                loadQuestion();
            }
        };

        loadQuestion();
    </script>
